Якщо файл unit_tests.cpp вже є — add_executable(...) створює тестовий виконуваний файл з цього джерела. Потрібно лише підключити GoogleTest (FetchContent або system package), включити потрібні інклуди та залінкувати GTest та вашу бібліотеку/сировини проекту, після чого додати add_test(...).

Ось компактний фрагмент CMake для вставки в ваш CMakeLists.txt (замінює або доповнює секцію тестів):

```cmake
// ...existing code...
option(BUILD_TESTS "Enable building of tests" ON)
if(BUILD_TESTS AND EXISTS "${CMAKE_SOURCE_DIR}/tests/unit_tests.cpp")
    enable_testing()

    include(FetchContent)
    FetchContent_Declare(
      googletest
      URL https://github.com/google/googletest/archive/refs/heads/main.zip
    )
    FetchContent_MakeAvailable(googletest)

    # Створити тестовий виконуваний файл з наявного файлу
    add_executable(${PROJECT_NAME}_tests
        tests/unit_tests.cpp
    )

    # Інклуди (щоб коректно підключити math_operations.h)
    target_include_directories(${PROJECT_NAME}_tests PRIVATE ${CMAKE_SOURCE_DIR})

    # Лінкування: GTest та (за наявності) бібліотека проекту
    if(TARGET ${PROJECT_NAME}_lib)
        target_link_libraries(${PROJECT_NAME}_tests PRIVATE GTest::gtest_main ${PROJECT_NAME}_lib)
    else()
        # Якщо немає окремої бібліотеки – можна підключити всі джерела проекту в тести
        file(GLOB_RECURSE ALL_SOURCES CONFIGURE_DEPENDS
            "${CMAKE_SOURCE_DIR}/src/*.cpp"
            "${CMAKE_SOURCE_DIR}/*.cpp"
        )
        if(ALL_SOURCES)
            target_sources(${PROJECT_NAME}_tests PRIVATE ${ALL_SOURCES})
        endif()
        target_link_libraries(${PROJECT_NAME}_tests PRIVATE GTest::gtest_main)
    endif()

    add_test(NAME unit_tests COMMAND ${PROJECT_NAME}_tests)
endif()
```
